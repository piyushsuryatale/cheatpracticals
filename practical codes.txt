// 3. Write a program to store the first year percentage of students in an array. Write function for
// sorting array of floating point numbers in ascending order using - Selection Sort and display

// top five scores

#include<iostream>
using namespace std;

void selection(float arr[],int n){
    for(int i=0;i<n-1;i++){
        int si=i;
        for(int j=i+1;j<n;j++){
            if(arr[j]<=arr[si]){
                si=j;
            }
        }
        swap(arr[i],arr[si]);
    }
}
void display(float arr[],int n){
    cout<<"student percentage in asending order"<<endl;
    for(int i=0;i<n;i++){
        cout<<arr[i]<<" ";
    }
    cout<<endl;
    cout<<"top five scores"<<endl;
    for(int i=n-1;i>=0 && i>=n-5;i--){
        cout<<arr[i]<<" ";
    }
    cout<<endl;
}

    for(int i=0;i<n;i++){
        cin>>arr[i];

    return 0;
    
}

        temp=temp->next;

        cout<<"member deleted"<<endl;
        return;

    prev->next=temp->next;

    if(temp==tail){

// seats in each row. Doubly linked list has to be maintained to keep track of free seats in rows.

// b) The booking can be cancelled.

#include<iostream>
using namespace std;

struct node{
    int seat;
    node* prev;
    node* next;

};

//add a free seat in sorted order
void insert(node* &head,int seat){
    node* newnode=new node();

    newnode->seat=seat;
    newnode->prev=NULL;
    newnode->next=NULL;

    if(head==NULL){
        head=newnode;
        return;

    }
    if(seat<head->seat){
        newnode->next=head;
        head->prev=newnode;
        head=newnode;
        return ;

    }
    node* temp=head;

    while(temp->next!=NULL && temp->next->seat<seat){
        temp=temp->next;

    }
    if(temp->seat==seat || (temp->next!=NULL && temp->next->seat==seat)){
        delete newnode;
        return;
    }
    newnode->next=temp->next;
    newnode->prev=temp;

    if(temp->next!=NULL){
        temp->next->prev=newnode;
    }
    temp->next=newnode;
}
//remove a booked seat from free seat list
void book (node* &head,int seat){
    node*temp=head;

    while(temp!=NULL && temp->seat!=seat){
        temp=temp->next;
    }
    if(temp==NULL){
        return;
    }
    if(temp==head){
        head=head->next;
    }
    if(temp->prev!=NULL){
        temp->prev->next=temp->next;
    }
    delete temp;
}

// 11. In any language program, most syntax errors occur due to unbalanced delimiters such as (), {},
// []. Write a program using a stack to check whether a given expression is well parenthesized or

// not.



#include<iostream>
#include<stack>
using namespace std;

class solution{

public:
    bool isvalid(string str){
        stack<char>st;

        for(int i=0;i<str.size();i++){
            if(str[i]=='(' || str[i]=='[' || str[i]=='{'){
                st.push(str[i]);
            }
            else{
                if(st.size()==0){
                    return false;
                }

                if((st.top()=='(' && str[i]==')') ||
                   (st.top()=='[' && str[i]==']') ||
                   (st.top()=='{' && str[i]=='}')){
                    st.pop();
                }
                else{
                    return false;
                }
            }
        }

        return st.size()==0;
    }
};

int main(){
    solution s;
    string str;

    cout<<"enter the statement "<<endl;
    cin>>str;

    if(s.isvalid(str)){
        cout<<"is valid"<<endl;
    }
    else{
        cout<<"not valid"<<endl;
    }

    return 0;
}



// 12. Pizza parlor accepting maximum M orders. Orders are served on a first-come, first-served
// basis. Queues are frequently used in computer programming, and a typical example is the
// creation of a job queue by an operating system. If the operating system does not use priorities,
// then the jobs are processed in the order they enter the system. Write a program for simulating a
// job queue. Write functions to add jobs and delete jobs from the queue.

#include<iostream>
using namespace std;

#define M 5

struct queue{
    int order[M];
    int front;
    int rear;
};

//add order
void insert(queue &q,int x){
    if(q.rear==M-1){
        cout<<"Queue is full"<<endl;
        return;
    }

    if(q.front==-1){
        q.front=0;
    }

    q.rear++;
    q.order[q.rear]=x;
}

//delete order
void del(queue &q){
    if(q.front==-1){
        cout<<"Queue is empty"<<endl;
        return;
    }

    cout<<"Deleted order: "<<q.order[q.front]<<endl;

    if(q.front==q.rear){
        q.front=-1;
        q.rear=-1;
    }
    else{
        q.front++;
    }
}

//display order
void display(queue q){
    if(q.front==-1){
        cout<<"Queue is empty"<<endl;
        return;
    }

    for(int i=q.front;i<=q.rear;i++){
        cout<<q.order[i]<<" ";
    }

    cout<<endl;
}

int main(){
    queue q;

    q.front=-1;
    q.rear=-1;

    int ch,x;

    while(1){
        cout<<"\n1. Insert order";
        cout<<"\n2. Delete order";
        cout<<"\n3. Display order";
        cout<<"\n4. Exit";

        cout<<"\nEnter choice: ";
        cin>>ch;

        if(ch==1){
            cout<<"Enter order number: ";
            cin>>x;

            insert(q,x);
        }
        else if(ch==2){
            del(q);
        }
        else if(ch==3){
            display(q);
        }
        else if(ch==4){
            return 0;
        }
        else{
            cout<<"Invalid choice"<<endl;
        }
    }

    return 0;
}



//  13 .Beginning with an empty binary search tree, Construct a binary search tree by inserting the

// values in the order given. After constructing a binary tree -
// 1. Insert new node.

// 2. Find number of nodes in the longest path from the root.
// 3. Minimum data value found in the tree


#include<iostream>
using namespace std;

struct node{
    int data;
    node* left;
    node* right;
};

//insert node
node* insert(node* root,int val){

    if(root==NULL){
        node* newnode=new node();
        newnode->data=val;
        newnode->left=NULL;
        newnode->right=NULL;
        return newnode;
    }

    if(val<root->data){
        root->left=insert(root->left,val);
    }
    else if(val>root->data){
        root->right=insert(root->right,val);
    }

    return root;
}

//find height (longest path)
int height(node* root){
    if(root==NULL){
        return 0;
    }

    int left=height(root->left);
    int right=height(root->right);

    if(left>right){
        return left+1;
    }
    else{
        return right+1;
    }
}

//find minimum value
int minimum(node* root){
    if(root==NULL){
        return -1;
    }

    while(root->left!=NULL){
        root=root->left;
    }

    return root->data;
}

//display inorder
void inorder(node* root){
    if(root==NULL){
        return;
    }

    inorder(root->left);
    cout<<root->data<<" ";
    inorder(root->right);
}

        if(ch==1){
            cout<<"Enter value: ";

            root=insert(root,val);
        }
        else if(ch==2){
            cout<<"Longest path: "<<height(root)<<endl;

//  14. Beginning with an empty binary search tree, Construct a binary search tree by inserting the

// values in the order given. After constructing a binary tree—

// 1. Insert new node

// 2. Change a tree so that the roles of the left and right pointers are swapped

// at every node.
// 3. Search for a value.

#include<iostream>
using namespace std;

struct node{
    int data;
    node* left;
    node* right;
};

//insert new node
node* insert(node* root,int val){

    if(root==NULL){
        node* newnode=new node();

        newnode->data=val;
        newnode->left=NULL;
        newnode->right=NULL;

        return newnode;
    }

    if(val<root->data){
        root->left=insert(root->left,val);
    }
    else if(val>root->data){
        root->right=insert(root->right,val);
    }

    return root;
}

//mirror tree
void mirror(node* root){

    if(root==NULL){
        return;
    }

    node* temp=root->left;
    root->left=root->right;
    root->right=temp;

    mirror(root->left);
    mirror(root->right);
}

//search value
bool search(node* root,int key){

    if(root==NULL){
        return false;
    }

    if(root->data==key){
        return true;
    }

    if(key<root->data){
        return search(root->left,key);
    }
    else{
        return search(root->right,key);
    }
}

//inorder traversal
void inorder(node* root){

    if(root==NULL){
        return;
    }

    inorder(root->left);
    cout<<root->data<<" ";
    inorder(root->right);
}

int main(){

    node* root=NULL;
    int n,val,ch;

    cout<<"Enter number of nodes: ";
    cin>>n;

    cout<<"Enter values: ";

    for(int i=0;i<n;i++){
        cin>>val;
        root=insert(root,val);
    }

    while(1){

        cout<<"\n1. Insert new node";
        cout<<"\n2. Mirror tree";
        cout<<"\n3. Search value";
        cout<<"\n4. Display tree";
        cout<<"\n5. Exit";

        cout<<"\nEnter choice: ";
        cin>>ch;

        if(ch==1){
            cout<<"Enter value: ";
            cin>>val;

            root=insert(root,val);
        }
        else if(ch==2){
            mirror(root);
            cout<<"Tree mirrored successfully"<<endl;
        }
        else if(ch==3){
            cout<<"Enter value to search: ";
            cin>>val;

            if(search(root,val)){
                cout<<"Value found"<<endl;
            }
            else{
                cout<<"Value not found"<<endl;
            }
        }
        else if(ch==4){
            cout<<"Inorder: ";
            inorder(root);
            cout<<endl;
        }
        else if(ch==5){
            return 0;
        }
        else{
            cout<<"Invalid choice"<<endl;
        }
    }

    return 0;
}
    


// 15. There are flight paths between cities. If there is a flight between city A and city B then there is
// an edge between the cities. The cost of the edge can be the time that flight takes to reach city B
// from A or the amount of fuel used for the journey. Represent this as a graph. The node can be
// represented by the airport name or name of the city. Use adjacency list representation of the

// graph .Check whether the graph is connected or not.



#include<iostream>
#include<vector>
using namespace std;

#define N 5

vector<pair<int,int>> adj[N];
bool visited[N];

void addEdge(int u,int v,int cost){
    adj[u].push_back({v,cost});
    adj[v].push_back({u,cost});
}

void DFS(int src){
    visited[src]=true;

    for(int i=0;i<adj[src].size();i++){
        int v=adj[src][i].first;

        if(visited[v]==false){
            DFS(v);
        }
    }
}

bool isConnected(){
    for(int i=0;i<N;i++){
        visited[i]=false;
    }

    DFS(0);

    for(int i=0;i<N;i++){
        if(visited[i]==false){
            return false;
        }
    }

    return true;
}

int main(){

    addEdge(0,1,2);
    addEdge(0,2,3);
    addEdge(1,3,4);
    addEdge(2,4,5);

    cout<<"Adjacency List:"<<endl;

    for(int i=0;i<N;i++){
        cout<<i<<" -> ";

        for(int j=0;j<adj[i].size();j++){
            cout<<"("<<adj[i][j].first<<", "
                <<adj[i][j].second<<") ";
        }

        cout<<endl;
    }

    if(isConnected()){
        cout<<"Graph is Connected";
    }
    else{
        cout<<"Graph is Not Connected";
    }

    return 0;
}
